## Part 4: Services & AI Logic

### AI Property Matching Service
```javascript
// Location: /services/aiMatchingService.js

calculatePropertyMatch(client, property) {
  let score = 0;
  let reasons = [];
  
  // Budget matching (0-30 points)
  if (client.budget_min && client.budget_max && property.price) {
    if (property.price >= client.budget_min && property.price <= client.budget_max) {
      score += 30;
      reasons.push("Budget match");
    } else if (property.price < client.budget_min) {
      score += 15; // Below budget but might be interested
      reasons.push("Below budget");
    } else if (property.price <= client.budget_max * 1.1) {
      score += 20; // Slightly above budget
      reasons.push("Near budget");
    }
  }
  
  // Property type matching (0-25 points)
  if (client.property_type_interested && property.property_type) {
    if (client.property_type_interested.includes(property.property_type)) {
      score += 25;
      reasons.push("Preferred property type");
    }
  }
  
  // Location matching (0-20 points)
  if (client.location_preferred && property.area) {
    if (client.location_preferred.toLowerCase().includes(property.area.toLowerCase())) {
      score += 20;
      reasons.push("Preferred location");
    }
  }
  
  // Bedroom/Bathroom matching (0-15 points)
  if (client.required_bedrooms && property.bedrooms >= client.required_bedrooms) {
    score += 15;
    reasons.push(`${property.bedrooms} bedrooms`);
  }
  
  // Amenities matching (0-10 points)
  if (client.required_amenities && property.amenities) {
    const propertyAmenities = property.amenities.split(',');
    const clientAmenities = client.required_amenities.split(',');
    const matchedAmenities = clientAmenities.filter(a => propertyAmenities.includes(a)).length;
    const amenityScore = (matchedAmenities / clientAmenities.length) * 10;
    score += amenityScore;
    if (matchedAmenities > 0) reasons.push(`${matchedAmenities} amenities match`);
  }
  
  // Determine confidence level
  let confidence = score < 30 ? 'low' : score < 70 ? 'medium' : 'high';
  
  return {
    match_score: Math.min(100, score),
    match_reasons: reasons.join(', '),
    confidence_level: confidence
  };
}

// Get suggestions for client
async getSuggestionsForClient(client_id) {
  const client = await Client.findById(client_id);
  const properties = await Property.find({ status: 'available' });
  
  const matches = properties
    .map(property => ({
      property,
      ...this.calculatePropertyMatch(client, property)
    }))
    .filter(match => match.match_score >= 50)
    .sort((a, b) => b.match_score - a.match_score)
    .slice(0, 10); // Top 10 matches
  
  return matches;
}
```

### WhatsApp AI Agent Service
```javascript
// Location: /services/whatsappAIService.js

async handleIncomingMessage(phone_number, message) {
  // Find or create client
  let client = await Client.findOne({ phone: phone_number });
  if (!client) {
    client = await Client.create({
      phone: phone_number,
      source: 'whatsapp',
      status: 'inquiry'
    });
  }
  
  // Check if blacklisted
  const isBlacklisted = await Blacklist.findOne({ phone_number, is_active: true });
  if (isBlacklisted) {
    return await this.sendMessage(phone_number, "Thank you for your interest. We'll contact you soon.");
  }
  
  // Classify message intent
  const intent = await this.classifyIntent(message);
  
  // Route to appropriate handler
  switch(intent) {
    case 'lead_qualification':
      return await this.qualifyLead(client, message);
    case 'property_inquiry':
      return await this.suggestProperties(client, message);
    case 'follow_up':
      return await this.handleFollowUp(client, message);
    case 'booking_request':
      return await this.handleBooking(client, message);
    default:
      return await this.escalateToHuman(client, message);
  }
}

async suggestProperties(client, requirements) {
  // Update client requirements from message
  const extracted = await this.extractRequirements(requirements);
  await Client.updateOne({ id: client.id }, extracted);
  
  // Get AI matches
  const matches = await aiMatchingService.getSuggestionsForClient(client.id);
  
  if (matches.length === 0) {
    return await this.sendMessage(client.phone, 
      "We don't have exact matches right now, but we're looking! We'll contact you when we find something.");
  }
  
  // Send top 3 matches
  const topMatches = matches.slice(0, 3);
  for (const match of topMatches) {
    const message = `🏠 ${match.property.address}\n` +
                    `💰 ₹${match.property.price}\n` +
                    `📐 ${match.property.bedrooms}BHK | ${match.property.carpet_area} sqft\n` +
                    `✅ Match: ${match.match_score}% - ${match.match_reasons}`;
    await this.sendMessage(client.phone, message);
  }
  
  return await this.sendMessage(client.phone, 
    "Reply with 'interested' for any of these, or 'see more' for options!");
}

async scheduleAutomaticFollowUp(client_id, days = 1) {

  await FollowUp.create({

    status: 'pending'
  });
  
  return { scheduled: true, follow_up_date: followUpDate };
}
```

```javascript
// Location: /services/aiFollowupAgent.js

async executeScheduledFollowups() {
  // Find all due follow-ups
  const dueFollowups = await FollowUp.find({
    follow_up_date: { $lte: new Date() },
    status: 'pending'
  })
  .populate('client_id')
  .populate('assigned_to');
  
  for (const followup of dueFollowups) {
    const client = followup.client_id;
    
    // Get updated suggestion
    const matches = await aiMatchingService.getSuggestionsForClient(client.id);
    
    if (matches.length > 0) {
      const message = `Hi ${client.name}! 👋\n\n` +
                      `We found some properties that match your requirements!\n\n` +
                      `🏠 ${matches[0].property.address}\n` +
                      `💰 Price: ₹${matches[0].property.price}\n` +
                      `✅ Match: ${matches[0].match_score}%\n\n` +
                      `Would you like to schedule a viewing?`;
      
      await whatsappService.sendMessage(client.phone, message);
      
      // Schedule next follow-up (3 days later)
      await this.scheduleNextFollowup(client.id, 3);
    }
    
    // Mark as done
    await FollowUp.updateOne({ id: followup.id }, { status: 'done', completed_at: new Date() });
  }
}
```

```javascript
async uploadFile(organization_id, file, file_type = 'document') {
  const integration = await GoogleDriveIntegration.findOne({ organization_id });
  
  if (!integration) throw new Error('Google Drive not integrated');
  
  // Refresh token if expired
  if (integration.token_expiry < new Date()) {
    const newTokens = await googleAuth.refreshToken(integration.google_refresh_token);
    await GoogleDriveIntegration.updateOne({ id: integration.id }, {
      google_access_token: newTokens.access_token,
      token_expiry: newTokens.expiry_date
    });
    integration.google_access_token = newTokens.access_token;
  }
  
  // Upload file to Google Drive
  const fileMetadata = {
    name: file.originalname,
    parents: [integration.google_drive_folder_id],
    mimeType: file.mimetype
  };
  
  const fileData = file.buffer;
  const response = await googleDrive.files.create({
    auth: googleAuth.getAuthClient(integration.google_access_token),
    resource: fileMetadata,
    media: { body: fileData },
    fields: 'id, webViewLink'
  });
  
  return {
    google_drive_id: response.data.id,
    file_url: response.data.webViewLink
  };
}
```
